DisplayStateController : (States ändern)

                             void processDisplayState(); // process current display state
void updateDisplayState(DisplayState newState);          // catch some special transitions when changing display state
void savePreviousState();                                // save current display state for later restoration
void restorePreviousState();                             // restore previously saved display state
void goToDefaultState();                                 // go to default display state, showing clock
void revokeDisplayDelay();                               // reset timeout for state transition purposes
void abortTest();                                        // abort ongoing test sequence and return to default state

// save current display state and switch to LED brightness or CCT change screen
void switchToLedChangeScreen(DisplayState newState);

void checkIfGoToStandby(); // go to standby between 23:00 and 7:00 when light strip is off
void waitForRtc();         // wait until first message from RTC is received

// helper functions for incrementing/decrementing digits like hours, minutes in various display states
void incrementNumber();
void decrementNumber();
